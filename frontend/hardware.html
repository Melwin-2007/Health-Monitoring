#define MQ7_PIN 34

// Reads CO ppm using calibrated power law formula
float readMq7CoPpm() {
  int raw = analogRead(MQ7_PIN);
  float v = (raw / 4095.0) * 3.3;
  float rs = ((3.3 - v) / v) * 10.0;
  return pow((rs / 12.0) / 19.32, 1.0 / -0.64);
}
